    Create public/js/chart-sync.js — an ES module with no DOM and no dependency on the LightweightCharts global, so it runs in both the browser and vitest:
    ```
    function isUsableRange(range) {
      return !!range && Number.isFinite(range.from) && Number.isFinite(range.to);
    }

    export function createRangeSync() {
      let syncing = false;
      function link(from, to) {
        return from.subscribeVisibleLogicalRangeChange((range) => {
          if (syncing || !isUsableRange(range)) return;
          syncing = true;
          try {
            to.setVisibleLogicalRange(range);
          } finally {
            syncing = false;
          }
        });
      }
      return { link, isSyncing: () => syncing };
    }
    ```
    - `link(from, to)` subscribes to `from.timeScale()`-shaped objects (duck typing): `subscribeVisibleLogicalRangeChange(fn)` and `setVisibleLogicalRange(range)`. charts.js passes the real `chart.timeScale()` instances; the test passes fakes implementing the same contract.
    - The shared `syncing` boolean is the re-entrancy guard (CHART-08): while a programmatic set is in flight, the target chart's synchronous notify-back (LWC fires range-change handlers synchronously inside `setVisibleLogicalRange`) is ignored. This prevents the mutual feedback loop, double-fire, and stack overflow. Rapid pan/zoom events arriving in sequence are each applied once — the last one wins, so charts cannot desync.
    - `isUsableRange` rejects `null` and non-finite from/to so boundary cases (start-of-history, NaN from an edge clamp) never enter the apply path or throw (SC3: "does not crash").
    - The module references ONLY logical-range APIs — `subscribeVisibleLogicalRangeChange` / `setVisibleLogicalRange` — never time-range APIs (CHART-08).
    - Expose `isSyncing()` for the test to assert the guard flips on during apply.

    Create public/js/chart-sync.test.ts (vitest; imports `./chart-sync.js`). A FakeTimeScale that reproduces LWC's synchronous notify-back is the critical piece:
    ```
    import { describe, expect, it } from 'vitest';
    import { createRangeSync } from './chart-sync.js';

    class FakeTimeScale {
      constructor() {
        this.handlers = [];
        this.range = null;
        this.applyCount = 0;
      }
      subscribeVisibleLogicalRangeChange(fn) {
        this.handlers.push(fn);
        return () => { this.handlers = this.handlers.filter((h) => h !== fn); };
      }
      setVisibleLogicalRange(range) {
        this.applyCount += 1;
        this.range = range;
        for (const h of [...this.handlers]) h(range); // LWC notifies synchronously
      }
      fire(range) {
        for (const h of [...this.handlers]) h(range);
      }
      getVisibleLogicalRange() { return this.range; }
    }
    ```
    Test cases:
    1. Forwarding: `const sync = createRangeSync(); const a = new FakeTimeScale(); const b = new FakeTimeScale(); sync.link(a, b); a.fire({ from: 5, to: 45 });` → `b.applyCount === 1`, `b.range` deep-equals `{ from: 5, to: 45 }` (exact logical range preserved).
    2. Re-entrancy loop prevention (bidirectional wiring, the desync/double-fire/crash case): `sync.link(a, b); sync.link(b, a); a.fire({ from: 5, to: 45 });` → `b.applyCount === 1`, `a.applyCount === 0` (the synchronous notify-back from B is swallowed — no loop, no double-apply, no stack overflow). Then `b.fire({ from: 10, to: 50 });` → `a.applyCount === 1`, `b.applyCount` still `1` (symmetric).
    3. Rapid-fire (SC3 rapid pan/zoom): with links wired both ways, `a.fire({ from: 0, to: 10 }); a.fire({ from: 1, to: 11 }); a.fire({ from: 2, to: 12 });` → `b.applyCount === 3`, `b.range` deep-equals `{ from: 2, to: 12 }` (each event applied exactly once, last wins, no crash).
    4. Null range ignored: `a.fire(null)` → `b.applyCount === 0`.
    5. Non-finite range ignored (data-boundary edge): `a.fire({ from: -Infinity, to: 50 })` and `a.fire({ from: NaN, to: 10 })` → `b.applyCount === 0` in both cases.
    6. Gap tolerance by construction (CHART-08): A has 200 bars, B has 150 bars (B's dataset has a 50-bar gap). `sync.link(a, b); a.fire({ from: 100, to: 149 });` → `b.range` deep-equals `{ from: 100, to: 149 }` forwarded verbatim as a logical range (bar indices), with no time conversion and no throw — demonstrating the sync never depends on identical time coverage. `expect(sync.isSyncing()).toBe(false)` after each fire (guard resets).
    7. Unsubscribe: capture `const un = sync.link(a, b); un(); a.fire({ from: 1, to: 2 });` → `b.applyCount === 0`.
    8. Exception-safety (W-1 fix): Create a FakeTimeScale that throws on setVisibleLogicalRange, then wire two-way links: `const sync = createRangeSync(); const a = new ThrowingTimeScale(); const b = new FakeTimeScale(); sync.link(a, b); b.fire({ from: 1, to: 2 });` → `expect(() => a.fire({ from: 1, to: 2 })).toThrow(); expect(sync.isSyncing()).toBe(false)` — verify that even after a throw, the guard resets and subsequent non-throwing ranges apply correctly.

    async function init() {
      const errorEl = document.getElementById('chart-error');
      try {
        const [btcRows, ethRows] = await Promise.all([loadWindow('BTCUSDT'), loadWindow('ETHUSDT')]);
        const btcChart = renderChart('btc-chart', btcRows.map(toCandle));
        const ethChart = renderChart('eth-chart', ethRows.map(toCandle));

        const btcScale = btcChart.chart.timeScale();
        const ethScale = ethChart.chart.timeScale();

        const initial = btcScale.getVisibleLogicalRange();
        if (initial) ethScale.setVisibleLogicalRange(initial);

        const sync = createRangeSync();
        sync.link(btcScale, ethScale);
        sync.link(ethScale, btcScale);
      } catch (error) {

      }
    }
